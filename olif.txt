WeddingPress Custom Nama Tamu (WDPCNT) adalah script JavaScript yang digunakan untuk mengganti teks "Tamu Undangan" pada elemen dengan class `namatamu` berdasarkan parameter yang ada di URL. 
Script ini mendukung beberapa parameter seperti `to`, `dear`, dan `kepada` untuk menyesuaikan nama penerima undangan. Serta dilengkapi dengan engine URL Normalizer untuk memperbaiki karakter khusus (&) secara otomatis.

	// Tujuan: Pastikan parameter selalu ter-encode dengan benar di address bar.
	// Ini mencegah masalah & yang memecah URL menjadi parameter berbeda.
	var rawSearch = location.search;
	if ( rawSearch.indexOf('?') !== -1 ) {
		var params = new URLSearchParams(rawSearch);
		var needsRedirect = false;
		
		for (const key of config.urlParams) {
		    if (rawSearch.indexOf('?' + key + '=') !== -1 || rawSearch.indexOf('&' + key + '=') !== -1) {

        		if ( rawName !== null ) {
        			// Decode dulu semua encoding yang mungkin sudah ada
        			var decoded = rawName
        				.replace(/&amp;/gi, '&')
        				.replace(/&#38;/gi, '&')
        				.replace(/&#x26;/gi, '&')
        				.replace(/&apos;/gi, "'")
        				.replace(/&#39;/gi, "'")

      var hash = window.location.hash || '';
      var params = new URLSearchParams(search);
      var raw = params.get('to') || params.get('kepada') || params.get('nama') || params.get('u');

      return null;

    });

    ['cui_card-', 'wdk_card-'].forEach(function(prefix) {
      document.querySelectorAll('.' + prefix + key + ' span:first-child').forEach(function(el) {
        el.textContent = val;
      });
    });
  }

  function muatKomentar() {
    if (!GAS_URL || GAS_URL.indexOf('GANTI') > -1) return;
    fetch(GAS_URL)
      .then(function (r) { return r.json(); })
      .then(function (data) {
        if (!data.success) return;
        var total  = data.total || 0;
        var linkEl = document.getElementById('cui-link-3460');
        if (linkEl) linkEl.innerHTML = '<span>' + total + '</span> Ucapan';
        var c = data.counts || {};
        updateCounter('hadir',       c.hadir       || 0);
        updateCounter('tidak_hadir', c.tidak_hadir || 0);
        updateCounter('masih_ragu',  c.masih_ragu  || 0);
        var ul = document.getElementById('cui-container-comment-3460');
        if (!ul) return;
        if (!data.comments || data.comments.length === 0) {
          ul.innerHTML = '<li style="list-style:none;text-align:center;color:#aaa;padding:24px 0;">Belum ada ucapan. Jadilah yang pertama!</li>';
        } else {
          ul.innerHTML = data.comments.map(renderKomentar).join('');
        }
        var panel = document.getElementById('cui-wrap-commnent-3460');
        if (panel) panel.style.display = 'block';
      })
      .catch(function (err) { console.warn('[GuestBook] Load error:', err); });
  }

  function handleSubmit(e) {
    e.preventDefault();
    e.stopPropagation();
    e.stopImmediatePropagation();
    var form      = document.getElementById('commentform-3460');
    if (!form) return;
    var nama      = (form.querySelector('[name="author"]')    || {value:''}).value.trim();
    var ucapan    = (form.querySelector('[name="comment"]')   || form.querySelector('textarea') || {value:''}).value.trim();
    var konfirmasi = (form.querySelector('[name="konfirmasi"]') || {value:''}).value;
    var elBtn     = form.querySelector('[type="submit"]');
    if (!nama)             { alert('Mohon isi nama Anda.'); return; }
    if (ucapan.length < 2) { alert('Ucapan minimal 2 karakter.'); return; }
    if (elBtn) { elBtn.disabled = true; elBtn.value = 'Mengirim...'; }
    var statusEl = document.getElementById('cui-comment-status-3460');
    if (!GAS_URL || GAS_URL.indexOf('GANTI') > -1) {
      alert('URL Apps Script belum diisi di template.');
      if (elBtn) { elBtn.disabled = false; elBtn.value = 'Kirim'; }
      return;
    }
    fetch(GAS_URL, {
      method  : 'POST',
      body    : JSON.stringify({ nama: nama, ucapan: ucapan, konfirmasi: konfirmasi }),
      headers : { 'Content-Type': 'text/plain' }
    })
    .then(function (r) { return r.json(); })
    .then(function (data) {
      if (data.success) {
        form.reset();
        if (statusEl) {
          statusEl.innerHTML = '<p style="text-align:center;color:#3D9A62;padding:10px;font-size:14px;">&#10003; ' + (data.message || 'Ucapan tersimpan!') + '</p>';
          setTimeout(function () { statusEl.innerHTML = ''; }, 5000);
        }
        muatKomentar();
      } else {
        alert('Gagal: ' + (data.message || 'Coba lagi.'));
      }
    })
    .catch(function (err) { alert('Koneksi gagal. Coba lagi.'); })
    .finally(function () { if (elBtn) { elBtn.disabled = false; elBtn.value = 'Kirim'; } });
  }

  function init() {
    if (window.CUI_WP) window.CUI_WP.ajaxurl = GAS_URL;
    if (window.WDK_WP) window.WDK_WP.ajaxurl = GAS_URL;
    if (window.cevar)  window.cevar.ajax_url  = GAS_URL;
    var form = document.getElementById('commentform-3460');
    if (form) form.addEventListener('submit', handleSubmit, true);
    muatKomentar();
    var panel = document.getElementById('cui-wrap-commnent-3460');
    if (panel) panel.style.display = 'block';
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
  setTimeout(init, 800);
  setTimeout(muatKomentar, 2000);
})();
</script>
	
<script>
/* Countdown Timer Otomatis Stabil (2 Digit Tanpa Kedip) */
(function() {
    function initCountdown() {
        var target = new Date("September 28, 2026 10:00:00").getTime();
        
        var elDays = document.querySelector('[data-days]');
        var elHours = document.querySelector('[data-hours]');
        var elMinutes = document.querySelector('[data-minutes]');
        var elSeconds = document.querySelector('[data-seconds]');
        
        function pad2(num) {
            var s = String(num);
            while (s.length < 2) s = "0" + s;
            return s;
        }

        function tick() {
            var now = new Date().getTime();
            var diff = target - now;
            if (diff < 0) diff = 0;

            var d = Math.floor(diff / (1000 * 60 * 60 * 24));
            var h = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            var m = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
            var s = Math.floor((diff % (1000 * 60)) / 1000);

            if (elDays) elDays.innerText = pad2(d);
            if (elHours) elHours.innerText = pad2(h);
            if (elMinutes) elMinutes.innerText = pad2(m);
            if (elSeconds) elSeconds.innerText = pad2(s);
        }

        tick();
        setInterval(tick, 1000);
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initCountdown);
    } else {
        initCountdown();
    }
})();
</script>
